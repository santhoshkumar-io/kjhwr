pandas==3.1.0

# Makes Python verify TLS against the operating system's trust store instead of
# OpenSSL's own bundle at /etc/ssl/cert.pem.
#
# Needed because this network runs a TLS-inspecting proxy (Netskope). Its root CA
# is installed in the macOS keychain, so curl and the system Python 3.9 trust it,
# but the uv-managed CPython 3.12 reads /etc/ssl/cert.pem, which does not contain
# it - so every HTTPS call failed with CERTIFICATE_VERIFY_FAILED.
#
# Both scripts import it defensively, so they still run where it is absent (for
# example a container, which has no such proxy in front of it).
truststore==0.10.5
